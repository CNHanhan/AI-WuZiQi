**落子候选点优化：**  
AI 不需要遍历 225 个交叉点，只需在已有棋子周围 1~2 格范围内搜索候选点，大幅减少搜索空间。

```cpp
void Board::getValidMoves(std::vector<Point>& moves, int range) {
    bool visited[BOARD_SIZE][BOARD_SIZE] = {false};
    for (int i = 0; i < BOARD_SIZE; ++i) {
        for (int j = 0; j < BOARD_SIZE; ++j) {
            if (grid[i][j] != 0) {
                for (int dx = -range; dx <= range; ++dx) {
                    for (int dy = -range; dy <= range; ++dy) {
                        int nx = i + dx, ny = j + dy;
                        if (isValid(nx, ny) && !visited[nx][ny] && grid[nx][ny] == 0) {
                            moves.push_back({nx, ny});
                            visited[nx][ny] = true;
                        }
                    }
                }
            }
        }
    }
}
```

### 3.3 局面评分函数

```cpp
int evaluateBoard(const Board& board, int aiColor) {
    int myScore = 0;
    int oppScore = 0;
    int oppColor = (aiColor == 1) ? 2 : 1;

    // 遍历棋盘上每个已有棋子，评估其形成的棋型
    for (int i = 0; i < BOARD_SIZE; ++i) {
        for (int j = 0; j < BOARD_SIZE; ++j) {
            if (board.grid[i][j] == aiColor) {
                myScore += evaluatePosition(board, i, j, aiColor, oppColor);
            } else if (board.grid[i][j] == oppColor) {
                oppScore += evaluatePosition(board, i, j, oppColor, aiColor);
            }
        }
    }

    // 进攻权重 > 防守权重（鼓励主动进攻）
    return myScore * 1.2f - oppScore;
}
```

### 4.1 简单AI：威胁检测 + 贪心

```cpp
Point EasyAI::getMove(const Board& board, int aiColor) {
    // 1. 检查AI是否有必胜落子点
    Point winMove = findWinningMove(board, aiColor);
    if (winMove.isValid()) return winMove;

    // 2. 检查玩家是否有必胜落子点（必须防守）
    int oppColor = (aiColor == 1) ? 2 : 1;
    Point blockMove = findWinningMove(board, oppColor);
    if (blockMove.isValid()) return blockMove;

    // 3. 随机选择一个候选点
    std::vector<Point> moves;
    board.getValidMoves(moves);
    int idx = rand() % moves.size();
    return moves[idx];
}
```

### 4.2 中等AI：Minimax + Alpha-Beta 剪枝

```cpp
// 负无穷 ~ 正无穷
const int INF = 1000000000;

int alphaBeta(Board& board, int depth, int alpha, int beta,
              int currentColor, int aiColor) {
    // 终止条件
    int winner = board.checkWinner();
    if (winner == aiColor)     return INF;
    if (winner == getOpponent(aiColor)) return -INF;
    if (depth == 0)            return evaluateBoard(board, aiColor);

    std::vector<Point> moves;
    board.getValidMoves(moves, 2);  // 搜索范围：已有棋子周围2格
    sortMoves(moves, board, currentColor);  // 走法排序（启发式优化）

    for (const Point& mv : moves) {
        board.placeStone(mv.x, mv.y, currentColor);
        int score = alphaBeta(board, depth - 1, alpha, beta,
                               getOpponent(currentColor), aiColor);
        board.removeStone(mv.x, mv.y);

        if (currentColor == aiColor) {
            if (score > alpha) alpha = score;
            if (alpha >= beta) break;  // Beta 剪枝
        } else {
            if (score < beta) beta = score;
            if (alpha >= beta) break;  // Alpha 剪枝
        }
    }

    return (currentColor == aiColor) ? alpha : beta;
}

Point MediumAI::getMove(const Board& board, int aiColor) {
    const int DEPTH = 6;  // 搜索深度6层
    Point bestMove = {-1, -1};
    int bestScore = -INF;

    std::vector<Point> moves;
    board.getValidMoves(moves, 2);
    sortMoves(moves, board, aiColor);

    for (const Point& mv : moves) {
        Board temp = board;  // 拷贝棋盘
        temp.placeStone(mv.x, mv.y, aiColor);
        int score = alphaBeta(temp, DEPTH - 1, -INF, INF,
                               getOpponent(aiColor), aiColor);
        if (score > bestScore) {
            bestScore = score;
            bestMove = mv;
        }
    }
    return bestMove;
}
```

### 4.3 困难AI：MCTS（蒙特卡洛树搜索）

```cpp
struct MCTSNode {
    Board  board;          // 当前局面
    int     color;          // 当前要落子的颜色
    MCTSNode* parent;
    std::vector<std::unique_ptr<MCTSNode>> children;
    int     visits;         // 访问次数
    float   wins;           // 累计胜率（对AI而言）
    bool    expanded;       // 是否已展开
};

class MCTS {
    int iterations_;        // 模拟次数（10000+）
    int aiColor_;

    // 四个MCTS阶段
    MCTSNode* select(MCTSNode* node);       // 选择
    void      expand(MCTSNode* node);        // 展开
    float     simulate(MCTSNode* node);      // 模拟
    void      backpropagate(MCTSNode* node, float result); // 回溯

    // UCB1 公式选择子节点
    MCTSNode* bestChild(MCTSNode* node, double c = 1.414);

public:
    Point getMove(const Board& board, int aiColor);
};

// UCB1 公式
double ucb1(MCTSNode* child, MCTSNode* parent) {
    if (child->visits == 0) return 1e9;  // 优先访问未访问节点
    return (child->wins / child->visits)
           + c * sqrt(log(parent->visits) / child->visits);
}
```

**MCTS 优势：**
- 不需要精确的评估函数
- 随着模拟次数增加，决策质量持续提升
- 适合五子棋这种"先手优势"明显的游戏

### 5.1 Zobrist 哈希（局面快速去重）

```cpp
// 每个棋盘位置每种颜色对应一个随机64位整数
uint64_t zobristTable[BOARD_SIZE][BOARD_SIZE][3];

uint64_t getBoardHash(const Board& board) {
    uint64_t hash = 0;
    for (int i = 0; i < BOARD_SIZE; ++i)
        for (int j = 0; j < BOARD_SIZE; ++j)
            if (board.grid[i][j] != 0)
                hash ^= zobristTable[i][j][board.grid[i][j]];
    return hash;
}
// 配合 Transposition Table 缓存已评估局面
```

### 5.2 迭代深化（Iterative Deepening）

```cpp
// 先从浅层搜索开始，逐步加深
// 浅层最佳走法可用于深层搜索的走法排序（提高剪枝效率）
Point iterativeDeepening(const Board& board, int aiColor, int maxDepth) {
    Point bestMove;
    for (int depth = 2; depth <= maxDepth; depth += 2) {
        Point mv = alphaBetaSearch(board, aiColor, depth);
        if (!isTimeout()) bestMove = mv;  // 未超时才接受结果
        else break;
    }
    return bestMove;
}
```
